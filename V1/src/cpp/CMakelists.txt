cmake_minimum_required(VERSION 3.23)

project(secureCommunication LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(EXTERNAL_DIR ${CMAKE_SOURCE_DIR}/../../external/cpp)

add_library(doctest INTERFACE)
target_include_directories(doctest INTERFACE ${EXTERNAL_DIR})
add_library(sodium INTERFACE)
target_include_directories(sodium INTERFACE ${EXTERNAL_DIR}/libsodium-win64/include)
target_link_libraries(sodium INTERFACE
    ${EXTERNAL_DIR}/libsodium-win64/lib/libsodium.dll.a
)

add_executable(test_main)
enable_testing()
add_test(NAME test_main COMMAND test_main)
target_sources(test_main PRIVATE
    tests/test_main.cpp
    tests/encryption_tests.cpp
    tests/types_tests.cpp
    tests/inputhelpers_tests.cpp
    encryption/encryptionService.cpp
    encryption/encryptionEngine.cpp
    helpers/inputhelpers.cpp
    database_communication/keymaterial_client.cpp
    database_communication/padallocator.cpp
    types.cpp
)
target_include_directories(test_main PRIVATE
    ${CMAKE_SOURCE_DIR}
    encryption
    helpers
    tests
    database_communication
)
target_link_libraries(test_main PRIVATE 
    doctest
    sodium
)
add_custom_command(TARGET test_main POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        ${EXTERNAL_DIR}/libsodium-win64/bin/libsodium-26.dll
        $<TARGET_FILE_DIR:test_main>
)
